# /codify

You are codifying patterns and fixing orchestration breakdowns from this session into a queue of remediation reports. Reports go to `<repo>/.codify-inbox/<TIMESTAMP>/`. A `Stop` hook then runs each report as a background `claude -p` -- but that is OUT OF YOUR SCOPE. Your job is analysis + dispatch + write phase only.

You produce reports. You do not implement them.

## Bootstrap (preprocess)

```!
set -e

REPO_ROOT=/Users/skunkworks/Documents/Work/Personal/claude-code-skills

# Prereqs
if ! test -d "$REPO_ROOT"; then
  echo "ABORT: marketplace repo not at $REPO_ROOT"
  exit 0
fi
if ! test -d "$HOME/.claude/plugins/cache/claude-plugins-official/plugin-dev"; then
  echo "ABORT: plugin-dev not installed. Run: /plugin install plugin-dev@claude-plugins-official"
  exit 0
fi
if ! test -d "$HOME/.claude/plugins/cache/claude-plugins-official/skill-creator"; then
  echo "ABORT: skill-creator not installed. Run: /plugin install skill-creator@claude-plugins-official"
  exit 0
fi

# Resolve transcript
ENCODED_CWD=$(pwd | sed 's:/:-:g')
TRANSCRIPT_PATH="$HOME/.claude/projects/$ENCODED_CWD/${CLAUDE_SESSION_ID}.jsonl"
if ! test -f "$TRANSCRIPT_PATH"; then
  TRANSCRIPT_PATH=$(ls -t "$HOME"/.claude/projects/*/${CLAUDE_SESSION_ID}.jsonl 2>/dev/null | head -1)
fi
if ! test -f "$TRANSCRIPT_PATH"; then
  echo "ABORT: transcript not found for session $CLAUDE_SESSION_ID"
  exit 0
fi

# Run dirs
TIMESTAMP=$(date +%Y%m%dT%H%M%S)
INBOX="$REPO_ROOT/.codify-inbox/$TIMESTAMP"
RUNS_DIR="$HOME/.claude/codify-runs/$TIMESTAMP"
SENTINEL_DIR="$HOME/.claude/codify"
SENTINEL="$SENTINEL_DIR/.codify-active-${CLAUDE_SESSION_ID}"
mkdir -p "$INBOX" "$RUNS_DIR" "$SENTINEL_DIR"

# Build dependency map from on-disk repo state (NOT from research dossier --
# the dossier may be stale; the filesystem is truth).
DEP_MAP="$INBOX/_dep-map.yaml"
EXISTING_SKILLS="$INBOX/_existing-skills.txt"

cd "$REPO_ROOT"
{
  echo "# Generated by /codify at $TIMESTAMP"
  echo "marketplace:"
  jq -r '"  name: " + .name + "\n  version: " + .metadata.version' .claude-plugin/marketplace.json
  echo "plugins:"
  for p in plugins/*/; do
    pname=$(basename "$p")
    test -f "$p/.claude-plugin/plugin.json" || continue
    pversion=$(jq -r '.version // "unknown"' "$p/.claude-plugin/plugin.json")
    echo "  - name: $pname"
    echo "    version: $pversion"
    echo "    path: $REPO_ROOT/$p"
    echo "    commands:"
    if test -d "$p/commands"; then
      for c in "$p"commands/*.md; do
        test -f "$c" || continue
        cname=$(basename "$c" .md)
        ahint=$(awk '/^---$/{f++; next} f==1' "$c" | grep -E '^argument-hint:' | head -1 | sed 's/argument-hint:[ ]*//')
        agents_used=$(awk '/^---$/{f++; next} f==1' "$c" | grep -oE 'Agent\([^)]+\)' | sed 's/Agent(//;s/)//' | sort -u | tr '\n' ',' | sed 's/,$//')
        echo "      - name: $cname"
        echo "        path: $REPO_ROOT/$c"
        test -n "$ahint" && echo "        argument_hint: $ahint"
        test -n "$agents_used" && echo "        invokes_agents: [$agents_used]"
      done
    fi
    echo "    agents:"
    if test -d "$p/agents"; then
      for a in "$p"agents/*.md; do
        test -f "$a" || continue
        aname=$(basename "$a" .md)
        skills_loaded=$(awk '/^skills:/{flag=1; next} flag && /^[^[:space:]-]/{flag=0} flag' "$a" | grep -E '^[ ]+-' | sed 's/^[ ]*-[ ]*//' | tr '\n' ',' | sed 's/,$//')
        echo "      - name: $aname"
        echo "        path: $REPO_ROOT/$a"
        test -n "$skills_loaded" && echo "        loads_skills: [$skills_loaded]"
      done
    fi
    echo "    skills:"
    if test -d "$p/skills"; then
      for s in "$p"skills/*/; do
        sname=$(basename "$s")
        case "$sname" in _*) continue;; esac
        test -f "$s/SKILL.md" || continue
        echo "      - name: $sname"
        echo "        path: $REPO_ROOT/$s/SKILL.md"
      done
    fi
  done
} > "$DEP_MAP"

{
  for d in plugins/*/skills/*/; do
    sname=$(basename "$d")
    case "$sname" in _*) continue;; esac
    pname=$(basename "$(dirname "$(dirname "$d")")")
    echo "$pname:$sname"
  done
  for f in plugins/*/commands/*.md; do
    test -f "$f" || continue
    cname=$(basename "$f" .md)
    pname=$(basename "$(dirname "$(dirname "$f")")")
    echo "$pname:$cname (command)"
  done
} | sort -u > "$EXISTING_SKILLS"

cat <<EOF
BOOTSTRAP_OK=1
REPO_ROOT=$REPO_ROOT
TRANSCRIPT_PATH=$TRANSCRIPT_PATH
TIMESTAMP=$TIMESTAMP
INBOX=$INBOX
RUNS_DIR=$RUNS_DIR
SENTINEL=$SENTINEL
DEP_MAP=$DEP_MAP
EXISTING_SKILLS=$EXISTING_SKILLS
DEP_MAP_LINES=$(wc -l < "$DEP_MAP" | tr -d ' ')
EXISTING_SKILLS_LINES=$(wc -l < "$EXISTING_SKILLS" | tr -d ' ')
EOF
```
